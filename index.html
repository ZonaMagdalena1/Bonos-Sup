function doPost(e){
  try{
    var ss=SpreadsheetApp.getActiveSpreadsheet();
    var sheet=ss.getSheetByName('BD_Bonos')||ss.getActiveSheet();
    var data;
    try{data=JSON.parse(e.postData.contents);}
    catch(ex){data=JSON.parse(e.parameter.payload||'{}');}
    if(data.action==='ping'){return out({ok:true});}
    if(data.action==='save'){
      var r=data.row;
      if(sheet.getLastRow()===0){
        sheet.appendRow(['ID','Fecha','Supervisor','Puesto','Semana','Rancho',
          'Contrato','Camioneta','Modelo','Placas','Evaluador','Jefe','Km',
          'Puntaje','Bono','Suspendido','Herr_Falta','Incumplimientos',
          'Negligencia','Fotos','Observaciones','Timestamp']);
      }
      sheet.appendRow([r.id,r.fecha,r.nombre,r.puesto,r.semana,r.rancho,
        r.contrato,r.cam,r.modelo,r.placas,r.evaluador,r.jefe,r.km,
        r.puntaje,r.bonoTxt,r.bonoTag==='suspendido'?'SI':'NO',
        r.herrFalta?'SI':'NO',r.incumplidos,r.negVal,r.nFotos,
        r.obsGeneral,new Date().toISOString()]);
      return out({ok:true});
    }
    return out({ok:false,msg:'unknown'});
  }catch(err){return out({ok:false,msg:err.toString()});}
}
function doGet(e){return out({ok:true,msg:'active'});}
function out(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);}
